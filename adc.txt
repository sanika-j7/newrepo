#include "stm32f407xx.h"
#include<stdio.h>
void ADC1_Init(void);
uint16_t ADC1_Read(void);
void USART2_Init(void);
void USART2_SendChar(char c);
void USART2_SendString(char *str);
void delay(void);
char buffer[50];
int main(void)
{
uint16_t ADC_value;
uint16_t temp;
ADC1_Init();
USART2_Init();
while(1)
{
ADC_value = ADC1_Read();
temp = (ADC_value * 330) / 4095;
sprintf(buffer, "Temp: %d C\r\n", temp);
USART2_SendString(buffer);
delay();
}
}
void ADC1_Init(void)
{
RCC -> AHB1ENR |= 1;
GPIOA -> MODER |= (3<<2);
RCC -> APB2ENR |= (1<<8);
ADC1 -> SMPR2 |= (7<<3);
ADC1 -> SQR3 = 1;
ADC1 -> CR2 = 1;
}
uint16_t ADC1_Read(void)
{
ADC1 -> CR2 |= (1<<30);
while(!(ADC1 -> SR && (1<<1)));
return ADC1 -> DR;
}
void USART2_Init(void){
RCC->AHB1ENR |= 1;
RCC->APB1ENR |= (1<<17);
GPIOA->MODER |= (2<<4);
GPIOA -> AFR[0] |= (7<<8);
USART2-> BRR = 0x0683;
USART2->CR1 |= (1<<3) | (1<<13);
}
void USART2_SendChar(char c){
while(!(USART2->SR & (1<<7)));
USART2->DR = c;
}
void USART2_SendString(char *str){
while(*str)
USART2_SendChar(*str++);
}
void delay(void)
{
for(volatile int i=0; i<500000; i++);
}
