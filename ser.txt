#include "stm32f407xx.h"
void SystemClock_8MHz(void);
void USART2_Init(void);
void USART2_SendChar(char c);
char USART2_ReceiveChar(void);
void USART2_SendString(char *str);
// ===================== MAIN =====================
int main(void)
{
char rx;
SystemClock_8MHz();
USART2_Init();
USART2_SendString("String is Transmitting:\r\n");
while(1)
{
rx = USART2_ReceiveChar();
USART2_SendChar(rx);
}
}
// ===================== SYSTEM CLOCK (HSE 8MHz) =====================
void SystemClock_8MHz(void)
{
RCC->CR |= RCC_CR_HSEON; // Enable HSE
while(!(RCC->CR & RCC_CR_HSERDY)); // Wait ready
RCC->CFGR &= ~RCC_CFGR_SW; // Clear clock switch
RCC->CFGR |= RCC_CFGR_SW_HSE; // Select HSE
while((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_HSE);
}
// ===================== USART2 INIT =====================
void USART2_Init(void)
{
// Enable clocks
RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
RCC->APB1ENR |= RCC_APB1ENR_USART2EN;
// PA2 (TX) and PA3 (RX) Alternate Function mode
GPIOA->MODER &= ~((3<<4) | (3<<6)); // Clear bits
GPIOA->MODER |= (2<<4) | (2<<6); // AF mode
// AF7 for USART2
GPIOA->AFR[0] &= ~((0xF<<8) | (0xF<<12));
GPIOA->AFR[0] |= (7<<8) | (7<<12);
// 9600 baud @ 8MHz
USART2->BRR = 0x0341; // 8MHz / 9600 = 833 ≈ 0x341
// Enable TX, RX, USART
USART2->CR1 = USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;
}
// ===================== SEND CHAR =====================
void USART2_SendChar(char c)
{
while(!(USART2->SR & USART_SR_TXE));
USART2->DR = c;
}
// ===================== RECEIVE CHAR =====================
char USART2_ReceiveChar(void)
{
while(!(USART2->SR & USART_SR_RXNE));
return (char)USART2->DR;
}
// ===================== SEND STRING =====================
void USART2_SendString(char *str)
{
while(*str)
USART2_SendChar(*str++);
}
