#include "stm32f407xx.h"
#include <string.h>
void SystemClock_8MHz(void);
void USART2_Init(void);
void USART2_SendChar(char c);
char USART2_ReceiveChar(void);
void USART2_SendString(char *str);
int main(void)
{
char rx;
char gps_buffer[120];
char utc_time[20];
char date[20];
char latitude[20];
char longitude[20];
int i = 0;
int field = 0;
int t_i, d_i, lat_i, lon_i;
SystemClock_8MHz();
USART2_Init();
USART2_SendString("GPRMC Time-Date Parser Ready\r\n");
while(1)
{
rx = USART2_ReceiveChar();
if(rx == '$')
{
i = 0;
// Read complete line
while(rx != '\n' && i < sizeof(gps_buffer)-1)
{
gps_buffer[i++] = rx;
rx = USART2_ReceiveChar();
}
gps_buffer[i] = '\0';
// Check GPRMC
if(strncmp(gps_buffer, "$GPRMC", 6) == 0)
{
field = 0;
t_i = d_i = lat_i = lon_i = 0;
for(i = 0; gps_buffer[i] != '\0'; i++)
{
if(gps_buffer[i] == ',')
{
field++;
continue;
}
// Field 1 → UTC Time
if(field == 1)
utc_time[t_i++] = gps_buffer[i];
// Field 3 → Latitude
if(field == 3)
latitude[lat_i++] = gps_buffer[i];
// Field 4 → N/S
if(field == 4)
latitude[lat_i++] = gps_buffer[i];
// Field 5 → Longitude
if(field == 5)
longitude[lon_i++] = gps_buffer[i];
// Field 6 → E/W
if(field == 6)
longitude[lon_i++] = gps_buffer[i];
// Field 9 → Date
if(field == 9)
date[d_i++] = gps_buffer[i];
}
utc_time[t_i] = '\0';
latitude[lat_i] = '\0';
longitude[lon_i] = '\0';
date[d_i] = '\0';
USART2_SendString("UTC Time: ");
USART2_SendString(utc_time);
USART2_SendString("\r\n");
USART2_SendString("Date: ");
USART2_SendString(date);
USART2_SendString("\r\n");
USART2_SendString("Latitude: ");
USART2_SendString(latitude);
USART2_SendString("\r\n");
USART2_SendString("Longitude: ");
USART2_SendString(longitude);
USART2_SendString("\r\n\r\n");
}
}
}
}
/* ================= CLOCK ================= */
void SystemClock_8MHz(void)
{
RCC->CR |= RCC_CR_HSEON;
while(!(RCC->CR & RCC_CR_HSERDY));
RCC->CFGR |= RCC_CFGR_SW_HSE;
while((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_HSE);
}
/* ================= USART2 ================= */
void USART2_Init(void)
{
RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;
RCC->APB1ENR |= RCC_APB1ENR_USART2EN;
// PA2 -> TX, PA3 -> RX
GPIOA->MODER &= ~((3<<4) | (3<<6));
GPIOA->MODER |= (2<<4) | (2<<6);
GPIOA->AFR[0] &= ~((0xF<<8) | (0xF<<12));
GPIOA->AFR[0] |= (7<<8) | (7<<12);
// 9600 baud @ 8MHz
USART2->BRR = 0x0341;
USART2->CR1 = USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;
}
/* ================= USART FUNCTIONS ================= */
void USART2_SendChar(char c)
{
while(!(USART2->SR & USART_SR_TXE));
USART2->DR = c;
}
char USART2_ReceiveChar(void)
{
while(!(USART2->SR & USART_SR_RXNE));
return USART2->DR;
}
void USART2_SendString(char *str)
{
while(*str)
USART2_SendChar(*str++);
}
